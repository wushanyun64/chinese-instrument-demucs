### Task 3: Fix `05_batch_inference.ipynb` — API update + space fixes

**File:** `notebooks/05_batch_inference.ipynb`

**Issues:**
1. Uses old `inference.separate.separate()` which now wraps `demucs.api.Separator` and only handles single files (no directory input)
2. Spaces in variable names

**Cell 2 (code)** — full rewrite:
```python
# Before:
from pathlib import Path
from inference.separate import separate
import torch

input_folder = Path('my_songs/')
output_folder = Path('target instrument_stems/')

if input_folder.exists():
    result = separate(
        input_path=input_folder, sig='<YOUR_SIG>',
        out_dir=output_folder,
        device='cuda' if torch.cuda.is_available() else 'cpu',
    )
    print(f'Instrument stems saved to: {result}')
else:
    print(f'Create {input_folder}/ with audio files to process')

# After:
from pathlib import Path
import demucs.api
import torch

input_folder = Path('my_songs/')
output_folder = Path('instrument_stems/')
sig = '<YOUR_SIG>'  # Replace with your trained model signature

if input_folder.exists():
    audio_files = sorted(
        p for p in input_folder.rglob('*')
        if p.suffix.lower() in {'.wav', '.flac', '.mp3', '.ogg', '.m4a'}
    )
    if not audio_files:
        print(f'No audio files found in {input_folder}/')
    else:
        separator = demucs.api.Separator(
            model=sig,
            device='cuda' if torch.cuda.is_available() else 'cpu',
        )
        for fp in audio_files:
            track_dir = output_folder / sig / fp.relative_to(input_folder).with_suffix('')
            track_dir.mkdir(parents=True, exist_ok=True)
            origin, separated = separator.separate_audio_file(fp)
            for stem_name, stem_wav in separated.items():
                demucs.api.save_audio(
                    stem_wav.cpu(),
                    str(track_dir / f'{stem_name}.wav'),
                    samplerate=separator.samplerate,
                )
            print(f'  {fp.name} → {track_dir}')
        print(f'Done — {len(audio_files)} files processed')
else:
    print(f'Create {input_folder}/ with audio files to process')
```

**Cell 4 (code)** — space fixes:

### Task 6: Final compilation check for all 6 notebooks

**Objective:** Verify every code cell in every notebook can be compiled (no syntax errors).

```bash
uv run python -c "
import json
from pathlib import Path

notebooks = sorted(Path('notebooks').glob('*.ipynb')) + sorted(Path('colab').glob('*.ipynb'))
errors = 0
for nb_path in notebooks:
    nb = json.loads(nb_path.read_text())
    for i, cell in enumerate(nb['cells']):
        if cell['cell_type'] == 'code':
            try:
                compile(''.join(cell['source']), f'{nb_path.name}:cell{i}', 'exec')
            except SyntaxError as e:
                print(f'SYNTAX ERROR: {nb_path.name} cell {i}: {e}')
                errors += 1
if errors:
    print(f'\n{errors} syntax error(s) found')
else:
    print('All code cells in all notebooks compile successfully')
"
```
